cmake_minimum_required(VERSION 3.16)
project(liujing_sensors)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs)
set(ORBBEC_SDK_ROOT "/home/ubuntu/sensor_bringup_20261021/SDK" CACHE PATH "Existing ARM64 Orbbec SDK")
add_executable(liujing_camera camera_ros2.cpp)
target_include_directories(liujing_camera PRIVATE ${ORBBEC_SDK_ROOT}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(liujing_camera ${ORBBEC_SDK_ROOT}/lib/arm64/libOrbbecSDK.so ${OpenCV_LIBS})
ament_target_dependencies(liujing_camera rclcpp sensor_msgs std_msgs)
install(TARGETS liujing_camera DESTINATION lib/${PROJECT_NAME})
ament_package()
