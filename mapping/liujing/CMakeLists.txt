cmake_minimum_required(VERSION 3.16)
project(liujing LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(liujing SHARED src/core.cpp hardware/raster.cpp)
target_include_directories(liujing PUBLIC src hardware)
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(liujing PRIVATE OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_options(liujing PRIVATE /O2 /fp:strict)
else()
  # Coverage and nearest-owner decisions require the same separately rounded
  # FP32 multiply/add operations as the HLS implementation. Keep O3/OpenMP.
  target_compile_options(liujing PRIVATE -O3 -ffp-contract=off -Wall -Wextra -Wno-unknown-pragmas)
endif()
add_executable(core_test tests/core_test.cpp)
target_link_libraries(core_test PRIVATE liujing)
enable_testing()
add_test(NAME geometry_and_raster COMMAND core_test)
